const CALENDARS = [
  'https://alice.btc.calendar.opentimestamps.org',
  'https://bob.btc.calendar.opentimestamps.org',
  'https://finney.calendar.eternitywall.com',
];

export type OtsResult = {
  ok: boolean;
  proof?: Buffer;
  calendars: string[];
  error?: string;
};

export async function submitToOts(hash: Uint8Array): Promise<OtsResult> {
  if (hash.length !== 32) {
    return { ok: false, calendars: [], error: 'hash_invalid_length' };
  }

  const successes: string[] = [];
  let proof: Buffer | undefined;

  for (const url of CALENDARS) {
    try {
      const res = await fetch(`${url}/digest`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/octet-stream' },
        body: Buffer.from(hash),
      });
      if (!res.ok) continue;

      const buf = Buffer.from(await res.arrayBuffer());
      const magic = Buffer.from([0x00,0x4f,0x70,0x65,0x6e,0x54,0x69,0x6d,0x65,0x73,0x74,0x61,0x6d,0x70,0x73,0x00,0x00,0x50,0x72,0x6f,0x6f,0x66,0x00]);
      if (!buf.subarray(0, magic.length).equals(magic)) continue;

      successes.push(url);
      if (!proof) proof = buf;
    } catch {}
  }

  if (!proof) {
    return { ok: false, calendars: successes, error: 'all_calendars_failed' };
  }
  return { ok: true, proof, calendars: successes };
}

export async function upgradeOts(hash: Uint8Array): Promise<Buffer | null> {
  const hashHex = Buffer.from(hash).toString('hex');
  for (const url of CALENDARS) {
    try {
      const res = await fetch(`${url}/timestamp/${hashHex}`);
      if (!res.ok) continue;
      const buf = Buffer.from(await res.arrayBuffer());
      if (buf.length > 0) return buf;
    } catch {}
  }
  return null;
}

export function generateCertificate(title: string, author: string, date: string, hash: string, qrCodeDataUrl: string) {
  const doc = new jsPDF({ orientation: 'landscape', unit: 'mm', format: 'a4' });
  doc.setFillColor(253, 248, 240); doc.rect(0, 0, 297, 210, 'F');
  doc.setDrawColor(231, 111, 81); doc.setLineWidth(2); doc.rect(10, 10, 277, 190);
  doc.setFont("helvetica", "bold"); doc.setFontSize(36); doc.setTextColor(44, 24, 16);
  doc.text("CERTIFICAT D'ANTÉRIORITÉ", 148.5, 40, { align: "center" });

  doc.setFontSize(14); doc.setTextColor(44, 24, 16);
  doc.text(`Titre : ${title}`, 30, 80); doc.text(`Créateur : @${author}`, 30, 90); doc.text(`Date : ${new Date(date).toLocaleString('fr-FR')}`, 30, 100);

export function generatePortfolioPDF(author: string, seals: any[], startDate?: string, endDate?: string, isPro: boolean = false) {
  const doc = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });
  
  doc.setFillColor(245, 233, 216); doc.rect(0, 0, 210, 297, 'F');
  doc.setFont("helvetica", "bold"); doc.setFontSize(40); doc.setTextColor(231, 111, 81);
  doc.text("PORTFOLIO", 105, 120, { align: "center" });
  doc.setFontSize(24); doc.setTextColor(44, 24, 16);
  doc.text(`@${author}`, 105, 140, { align: "center" });
  doc.setFontSize(12); doc.setTextColor(122, 90, 58);
  
  let dateText = `Généré le ${new Date().toLocaleDateString('fr-FR')}`;
  if (startDate && endDate) dateText += ` | Période : du ${new Date(startDate).toLocaleDateString('fr-FR')} au ${new Date(endDate).toLocaleDateString('fr-FR')}`;
  
  doc.text(dateText, 105, 155, { align: "center" });
  doc.text(`${seals.length} œuvre(s) certifiée(s)`, 105, 165, { align: "center" });

  doc.addPage();
  let y = 20;
  
  seals.forEach((seal, index) => {
    if (y > 240) { doc.addPage(); y = 20; }
    doc.setDrawColor(239, 224, 202); doc.setLineWidth(1); doc.rect(15, y, 180, 60);
    doc.setFont("helvetica", "bold"); doc.setFontSize(16); doc.setTextColor(44, 24, 16);
    doc.text(seal.title, 20, y + 10);
    doc.setFont("helvetica", "normal"); doc.setFontSize(10); doc.setTextColor(122, 90, 58);
    const category = seal.category || 'Art';
    const dateStr = new Date(seal.timestamp).toLocaleString('fr-FR');
    doc.text(`Catégorie : ${category} | Scellé le : ${dateStr}`, 20, y + 17);
    doc.text(`❤️ ${seal.likes_count || 0} J'aime   |   💬 ${seal.comments_count || 0} Commentaires`, 20, y + 23);
    if (seal.description) {
      doc.setFont("helvetica", "italic");
      const splitDesc = doc.splitTextToSize(`"${seal.description}"`, 170);
      doc.text(splitDesc, 20, y + 30);
    }
    doc.setFont("courier", "normal"); doc.setFontSize(8); doc.setTextColor(150, 150, 150);
    doc.text(`Hash : ${seal.hash ? seal.hash.slice(0, 80) + '...' : 'Non disponible'}`, 20, y + 50);
    if (seal.is_hosted && seal.file_path) {
      doc.setTextColor(74, 144, 194);
      doc.textWithLink(`[ Cliquez ici pour ouvrir l'œuvre originale ]`, 130, y + 50, { url: seal.file_path });
    } else if (seal.is_flash) {
       doc.setTextColor(196, 74, 46);
       doc.text(`[ ⚡ Sceau Flash - Fichier local ]`, 130, y + 50);
    }
    y += 70;
  });

  if (!isPro) {
    const pageCount = doc.getNumberOfPages();
    for (let i = 1; i <= pageCount; i++) {
      doc.setPage(i);
      doc.setFontSize(60);
      doc.setTextColor(200, 180, 160);
      doc.text("SCEAU FREE", 50, 150, { angle: 45 });
      doc.text("Passez PRO pour retirer", 50, 170, { angle: 45 });
    }
  }

  doc.save(`Portfolio-${author}.pdf`);
}

'use client';
import './globals.css'
import Link from 'next/link'
import { useState, useEffect } from 'react'
import { usePathname } from 'next/navigation'
import Manifeste from '@/components/Manifeste'

export default function RootLayout({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const [handle, setHandle] = useState<string | null>(null);
  const [isDark, setIsDark] = useState(false);
  
  const [notifications, setNotifications] = useState<any[]>([]);
  const [unread, setUnread] = useState(0);
  const [showNotifs, setShowNotifs] = useState(false);

  useEffect(() => {
    setHandle(localStorage.getItem('sceau_handle'));
    const theme = localStorage.getItem('sceau_theme');
    if (theme === 'dark') { setIsDark(true); document.documentElement.classList.add('dark'); }
  }, [pathname]);

  useEffect(() => {
    if (!handle) return;
    const fetchNotifs = () => {
      fetch(`/api/notifications?me=${handle}`).then(r => r.json()).then(d => {
        setNotifications(d.notifications || []);
        setUnread(d.unread || 0);
      });
    };
    fetchNotifs();
    const interval = setInterval(fetchNotifs, 10000);
    return () => clearInterval(interval);
  }, [handle]);

  const toggleTheme = () => {
    const next = !isDark; setIsDark(next);
    if (next) { document.documentElement.classList.add('dark'); localStorage.setItem('sceau_theme', 'dark'); }
    else { document.documentElement.classList.remove('dark'); localStorage.setItem('sceau_theme', 'light'); }
  };

  const markAsRead = () => {
    if (unread > 0 && handle) {
      fetch('/api/notifications/read', { method: 'POST', body: JSON.stringify({ me: handle }) });
      setUnread(0);
    }
  };

  const isActive = (path: string) => pathname === path;
  const isToolsActive = pathname === '/seal' || pathname === '/verify';

  // NOUVEAU : On définit sur quelles pages le Manifeste doit s'afficher
  const showManifeste = ['/profil', '/seal', '/verify', '/settings'].includes(pathname);

  return (
    <html lang="fr">
      <body className="min-h-screen pb-24">
        <nav className="bg-[var(--bg-surface)] border-b border-[var(--border-color)] px-6 py-4 flex justify-between items-center sticky top-0 z-50">
          <Link href="/feed" className="text-2xl font-bold tracking-tight font-serif" style={{color: 'var(--text-main)'}}>SCEAU<span className="text-[#E76F51]">.</span></Link>
          <div className="flex items-center gap-4">
            <button onClick={toggleTheme} className="text-xl cursor-pointer">{isDark ? '☀️' : '🌙'}</button>
            
            {handle && (
              <div className="relative">
                <button onClick={() => { setShowNotifs(!showNotifs); if (!showNotifs) markAsRead(); }} className="text-xl relative">
                  🔔
                  {unread > 0 && <span className="absolute -top-1 -right-1 bg-red-500 text-white text-[0.6rem] font-bold w-4 h-4 flex items-center justify-center rounded-full">{unread}</span>}
                </button>
                
                {showNotifs && (
                  <div className="absolute right-0 mt-4 w-72 bg-[var(--bg-surface)] border border-[var(--border-color)] rounded-2xl shadow-xl overflow-hidden z-50">
                    <div className="p-3 border-b border-[var(--border-color)] font-bold text-sm">Notifications</div>
                    <div className="max-h-64 overflow-y-auto">
                      {notifications.length === 0 ? <p className="p-4 text-center text-xs opacity-50">Aucune notification.</p> : 
                        notifications.map((n: any) => (
                          <Link key={n.id} href={n.link} onClick={() => setShowNotifs(false)} className={`block p-3 border-b border-[var(--border-color)] hover:bg-[var(--bg-main)] transition ${!n.is_read ? 'bg-[#FDF4F0]' : ''}`}>
                            <p className="text-xs" style={{color: 'var(--text-main)'}}><span className="font-bold text-[#E76F51]">@{n.actor}</span> {n.message}</p>
                            <span className="text-[0.6rem] font-mono text-[var(--text-muted)]">{new Date(n.created_at).toLocaleTimeString('fr-FR')}</span>
                          </Link>
                        ))
                      }
                    </div>
                  </div>
                )}
              </div>
            )}

            {!handle ? (
              <div className="flex items-center gap-3">
                <Link href="/pricing" className="text-sm font-bold text-[var(--text-muted)] hover:text-[var(--text-main)] transition hidden sm:block">Tarifs</Link>
                <Link href="/signup" className="text-sm font-bold bg-[#E76F51] text-white px-4 py-2 rounded-lg">Créer un compte</Link>
              </div>
            ) : (
              <Link href="/profil" className="text-sm font-bold" style={{color: 'var(--text-main)'}}>@{handle}</Link>
            )}
          </div>
        </nav>
        
        <main className="max-w-2xl mx-auto">{children}</main>
        
        {/* LE MANIFESTE S'AFFICHE CONDITIONNELLEMENT ICI */}
        {showManifeste && <Manifeste />}

        <div className="fixed bottom-0 left-0 right-0 bg-[var(--bg-surface)] border-t border-[var(--border-color)] flex justify-around py-3 z-50">
          <Link href="/feed" className={`flex flex-col items-center transition ${isActive('/feed') ? 'text-[#E76F51]' : 'text-[var(--text-muted)] hover:text-[#E76F51]'}`}>
            <span className="text-xl">🌐</span><span className="text-[0.6rem] font-bold font-mono mt-1">Feed</span>
          </Link>

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';
import crypto from 'crypto';
import { Resend } from 'resend';

export async function POST(req: Request) {
  const { email } = await req.json();
  if (!email) return NextResponse.json({ error: 'Email requis' }, { status: 400 });

  const user = await pool.query("SELECT id FROM users WHERE email = $1", [email]);
  if (user.rows.length === 0) return NextResponse.json({ error: 'Email introuvable' }, { status: 404 });

  const token = crypto.randomBytes(32).toString('hex');
  const expiry = new Date(Date.now() + 3600000); 

  await pool.query("UPDATE users SET reset_token = $1, reset_expiry = $2 WHERE email = $3", [token, expiry, email]);

  const resetLink = `${process.env.NEXT_PUBLIC_APP_URL || 'http://localhost:3000'}/reset-password?token=${token}`;
  
  try {
    const resendKey = process.env.RESEND_API_KEY;
    if (!resendKey) {
      // Si pas de clé, on simule l'envoi pour le développement
      console.log("⚠️ Pas de clé Resend. Lien de test :", resetLink);
      return NextResponse.json({ success: true, message: "Lien généré (voir console)" });
    }

    const resend = new Resend(resendKey);
    await resend.emails.send({
      from: 'SCEAU <onboarding@resend.dev>',
      to: email,
      subject: "Réinitialisation de votre mot de passe",
      html: `<p>Bonjour,</p><p>Voici votre lien : <a href="${resetLink}">Réinitialiser mon mot de passe</a></p>`,
    });
    return NextResponse.json({ success: true, message: "Email envoyé !" });
  } catch (error) {
    return NextResponse.json({ error: "Erreur d'envoi d'email" }, { status: 500 });
  }
}

    const { email, password } = await req.json();

    const user = await pool.query("SELECT * FROM users WHERE email = $1", [email]);
    if (user.rows.length === 0) {
      return NextResponse.json({ error: 'Email ou mot de passe incorrect' }, { status: 401 });
    }

    const isValid = await bcrypt.compare(password, user.rows[0].password_hash);
    if (!isValid) {
      return NextResponse.json({ error: 'Email ou mot de passe incorrect' }, { status: 401 });
    }

    // On renvoie le handle et la clé publique (convertie en base64)
    return NextResponse.json({
      handle: user.rows[0].handle,
      pubkey: Buffer.from(user.rows[0].pubkey).toString('base64')
    });
  } catch (error) {
    return NextResponse.json({ error: 'Erreur serveur' }, { status: 500 });
  }
}

export async function POST(req: Request) {
  const { token, newPassword } = await req.json();
  if (!token || !newPassword) return NextResponse.json({ error: 'Champs requis' }, { status: 400 });

  const user = await pool.query("SELECT id FROM users WHERE reset_token = $1 AND reset_expiry > NOW()", [token]);
  if (user.rows.length === 0) return NextResponse.json({ error: 'Lien invalide ou expiré' }, { status: 400 });

  const salt = await bcrypt.genSalt(10);
  const password_hash = await bcrypt.hash(newPassword, salt);

  await pool.query("UPDATE users SET password_hash = $1, reset_token = NULL, reset_expiry = NULL WHERE id = $2", [password_hash, user.rows[0].id]);

  return NextResponse.json({ success: true });
}

      return NextResponse.json({ error: 'Tous les champs sont requis' }, { status: 400 });
    }

    const salt = await bcrypt.genSalt(10);
    const password_hash = await bcrypt.hash(password, salt);

    const result = await pool.query(
      "INSERT INTO users (handle, email, password_hash, pubkey) VALUES ($1, $2, $3, decode($4, 'base64')) RETURNING id, handle", 
      [handle, email, password_hash, pubkey]
    );

  } catch (error: any) {
    console.error("Erreur d'inscription:", error);
    if (error.code === '23505') {
      return NextResponse.json({ error: 'Ce pseudo ou cet email est déjà pris' }, { status: 409 });
    }
    // On renvoie le VRAI message d'erreur pour le débogage
    return NextResponse.json({ error: error.message || 'Erreur serveur' }, { status: 500 });
  }

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';
import { sha256 } from '@noble/hashes/sha256';

export async function POST(req: Request) {
  try {
    const { pubkey, github_username } = await req.json();
    
    // 1. Sauvegarder le pseudo GitHub
    await pool.query("UPDATE users SET github_username = $1 WHERE encode(pubkey, 'base64') = $2", [github_username, pubkey]);
    const user = await pool.query("SELECT id FROM users WHERE encode(pubkey, 'base64') = $1", [pubkey]);
    const userId = user.rows[0].id;

    // 2. Récupérer les derniers commits publics via l'API GitHub
    const ghRes = await fetch(`https://api.github.com/users/${github_username}/events/public`, {
      headers: { 'User-Agent': 'SCEAU-App' }
    });
    const events = await ghRes.json();
    
    let commitsScelles = 0;
    for (const event of events) {
      if (event.type === 'PushEvent' && event.payload.commits.length > 0) {
        for (const commit of event.payload.commits) {
          // On crée un hash basé sur le message et l'ID du commit
          const hashStr = `${commit.sha}-${commit.message}`;
          const hash = sha256(new TextEncoder().encode(hashStr));
          let hashB64 = btoa(String.fromCharCode(...hash));

          // On insère dans la base (ON CONFLICT DO NOTHING pour éviter les doublons)
          await pool.query(
            `INSERT INTO seals (hash, timestamp, author_id, title, category, work_type, is_public) 
             VALUES (decode($1, 'base64'), $2, $3, $4, 'Code', 'commit', true) ON CONFLICT DO NOTHING`,
            [hashB64, event.created_at, userId, `Commit: ${commit.message.slice(0, 50)}`]
          );
          commitsScelles++;
        }
      }
    }
    return NextResponse.json({ success: true, commitsScelles });
  } catch (error) {
    return NextResponse.json({ error: 'Erreur GitHub' }, { status: 500 });
  }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';
import { sha256 } from '@noble/hashes/sha256';
import { submitToOts } from '@/lib/ots';

export async function POST(req: Request) {
  const { pact_id, pubkey } = await req.json();
  const user = await pool.query("SELECT id FROM users WHERE encode(pubkey, 'base64') = $1", [pubkey]);
  if (user.rows.length === 0) return NextResponse.json({ error: 'Inconnu' }, { status: 401 });

  try {
    await pool.query(
      "INSERT INTO pact_signatures (pact_id, user_id) VALUES ($1, $2) ON CONFLICT DO NOTHING",
      [pact_id, user.rows[0].id]
    );

    const pact = await pool.query(
      "SELECT title, description, required_signatures, status FROM pacts WHERE id = $1",
      [pact_id]
    );
    const sigs = await pool.query(
      `SELECT u.handle, s.created_at FROM pact_signatures s JOIN users u ON s.user_id = u.id
       WHERE s.pact_id = $1 ORDER BY s.created_at ASC`,
      [pact_id]
    );

    const required = pact.rows[0].required_signatures;
    const current = sigs.rows.length;

    if (current >= required && pact.rows[0].status !== 'scelle') {
      const canonical = JSON.stringify({
        title: pact.rows[0].title,
        description: pact.rows[0].description || '',
        signers: sigs.rows.map((r: any) => ({ handle: r.handle, signed_at: new Date(r.created_at).toISOString() })),
      });
      const pactHash = sha256(new TextEncoder().encode(canonical));

      let otsProof: string | null = null;
      try {
        const ots = await submitToOts(new Uint8Array(pactHash));
        if (ots.ok && ots.proof) otsProof = ots.proof.toString('base64');
      } catch (e) { console.error('[Pact OTS]', e); }

      await pool.query(
        `UPDATE pacts SET status='scelle', scelle_at=NOW(), pact_hash=$1, ots_proof=$2 WHERE id=$3`,
        [Buffer.from(pactHash), otsProof, pact_id]
      );
    }

    return NextResponse.json({ success: true, current, required });
  } catch (e) {
    console.error(e);
    return NextResponse.json({ error: 'Erreur' }, { status: 500 });
  }
}

    const userId = user.rows[0].id;
    const isPro = user.rows[0].is_pro || false;

    const stats = await pool.query(`
      SELECT COUNT(id) as total_sealed, SUM(CASE WHEN is_hosted = true THEN 1 ELSE 0 END) as total_hosted
      FROM seals WHERE author_id = $1
    `, [userId]);

    const seals = await pool.query(`
      SELECT s.id, s.title, s.description, s.timestamp, s.visibility, s.is_flash, s.is_hosted, s.file_path, s.category, encode(s.hash, 'base64') as hash,
      (SELECT COUNT(*) FROM likes WHERE seal_id = s.id) as likes_count,

      FROM seals s WHERE s.author_id = $1 ORDER BY s.created_at DESC
    `, [userId]);

    const totalPacts = parseInt(pacts.rows[0].total_pacts || 0);

      seals.rows.map((s: any) => new Date(s.timestamp).toISOString().split('T')[0])
    );

    let cursor = new Date();
    cursor.setUTCHours(0, 0, 0, 0);
    if (!uniqueDatesSet.has(cursor.toISOString().split('T')[0])) {
      cursor.setDate(cursor.getDate() - 1);
    }
    while (uniqueDatesSet.has(cursor.toISOString().split('T')[0])) {
      streak++;
      cursor.setDate(cursor.getDate() - 1);
    }

  const params = useParams();
  const [data, setData] = useState<any>(null);
  const [error, setError] = useState('');
  const myPubkey = typeof window !== 'undefined' ? localStorage.getItem('sceau_pubkey') : null;

  const load = () => {
    if (params.id) {
      fetch(`/api/challenges/${params.id}`).then(r => r.json()).then(d => {
        if (d.error) setError(d.error); else setData(d);
      });
    }
  };

  useEffect(() => { load(); }, [params.id]);

  async function vote(entryId: string) {
    if (!myPubkey) return alert("Connecte-toi pour voter !");
    const res = await fetch('/api/challenges/vote', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ challenge_id: params.id, entry_id: entryId, pubkey: myPubkey })
    });
    const d = await res.json();
    if (!res.ok) alert(d.error);
    else { alert("Vote enregistré !"); load(); }
  }

  if (error) return <div className="p-10 text-center text-red-500 font-bold">{error}</div>;
  if (!data) return <div className="p-10 text-center font-mono">Chargement du défi...</div>;

  return (
    <div className="p-6 max-w-2xl mx-auto font-serif">
      <Link href="/challenges" className="text-sm font-bold text-[#A89070] mb-4 inline-block hover:underline">← Retour aux défis</Link>

        <div className="mt-2 text-xs font-mono font-bold text-[#E76F51]">
          Fin le {new Date(data.challenge.deadline).toLocaleDateString('fr-FR')}

      </div>

  const [showPassword, setShowPassword] = useState(false);
  const [status, setStatus] = useState('');

  async function handleLogin() {
    if (!email || !password) return setStatus("Email et mot de passe requis");
    setStatus("Connexion...");
    
    const res = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password }) });
    const data = await res.json();
    
    if (res.ok) {
      localStorage.setItem('sceau_pubkey', data.pubkey);
      localStorage.setItem('sceau_handle', data.handle);
      setStatus("✅ Connecté !");
      setTimeout(() => router.push('/feed'), 500);
    } else {
      setStatus("❌ " + data.error);
    }
  }

  // La fonction magique pour "Entrée"
  const handleKeyPress = (e: React.KeyboardEvent) => { if (e.key === 'Enter') handleLogin(); };

  return (
    <div className="max-w-md mx-auto bg-[var(--bg-surface)] p-8 rounded-3xl border border-[var(--border-color)] shadow-sm mt-10">

  const [myHandle, setMyHandle] = useState('');
  const [contacts, setContacts] = useState<any[]>([]);
  const [groups, setGroups] = useState<any[]>([]);
  
  const [activeChat, setActiveChat] = useState<any>(null);
  const [messages, setMessages] = useState<any[]>([]);
  const [newMessage, setNewMessage] = useState('');
  
  const [showGroupModal, setShowGroupModal] = useState(false);
  const [groupName, setGroupName] = useState('');
  const [activePact, setActivePact] = useState<any>(null);

  const messagesEndRef = useRef<HTMLDivElement>(null);
  const prevMessagesLength = useRef(0);
  const myPubkey = typeof window !== 'undefined' ? localStorage.getItem('sceau_pubkey') : null;

  useEffect(() => {
    setMyHandle(localStorage.getItem('sceau_handle') || '');
    fetch('/api/messages').then(r => r.json()).then(d => setContacts(d.contacts || []));
    fetch(`/api/groups/list?me=${localStorage.getItem('sceau_handle')}`).then(r => r.json()).then(d => setGroups(d.groups || []));
  }, []);

  // Charger les messages avec FUSION intelligente
  useEffect(() => {
    if (!activeChat || !myHandle) return;
    const load = () => {
      const endpoint = activeChat.type === 'dm' 
        ? `/api/messages/${activeChat.id}?me=${myHandle}` 
        : `/api/groups/messages?id=${activeChat.id}`;

      fetch(endpoint).then(r => r.json()).then(data => {
        const serverMessages = data.messages || [];
        
        setMessages(prev => {
          // On garde les messages temporaires qui ne sont pas encore arrivés du serveur
          const tempMessages = prev.filter(m => m.isTemp);
          
          // On fusionne : les messages du serveur + les messages temporaires qui n'ont pas encore de correspondance
          const merged = [...serverMessages];
          
          tempMessages.forEach(temp => {
            // Si le message temporaire n'existe pas déjà dans la réponse serveur (comparaison de contenu et auteur)
            const isAlreadySaved = serverMessages.some(
              (sm: any) => sm.content === temp.content && sm.sender === temp.sender
            );
            if (!isAlreadySaved) {
              merged.push(temp);
            }
          });

          return merged;
        });
      });
    };
    
    load();
    const interval = setInterval(load, 3000);
    return () => clearInterval(interval);
  }, [activeChat, myHandle]);

  useEffect(() => { prevMessagesLength.current = 0; }, [activeChat]);

  useEffect(() => { 
    if (messages.length > prevMessagesLength.current) {
      messagesEndRef.current?.scrollIntoView({ behavior: "smooth" }); 
      prevMessagesLength.current = messages.length;
    }
  }, [messages]);

  async function sendMessage() {
    if (!newMessage.trim() || !activeChat) return;
    const msg = newMessage;
    setNewMessage('');
    
    // On ajoute un ID temporaire unique pour le suivi
    const tempId = `temp-${Date.now()}`;
    setMessages(prev => [...prev, { content: msg, sender: myHandle, created_at: new Date().toISOString(), isTemp: true, id: tempId }]);

    try {
      if (activeChat.type === 'dm') {
        await fetch('/api/messages', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ sender_handle: myHandle, receiver_handle: activeChat.id, content: msg }) });
      } else {
        await fetch('/api/groups/messages', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ group_id: activeChat.id, sender_handle: myHandle, content: msg }) });
      }
      // Une fois le message envoyé avec succès, on force un rechargement immédiat
      // pour remplacer le message temporaire par le vrai message du serveur
      setTimeout(() => {
        setMessages(prev => prev.filter(m => m.id !== tempId));
      }, 500);
    } catch (e) {
      console.error("Erreur d'envoi", e);
    }
  }

  async function createGroup() {
    if (!groupName) return;
    const res = await fetch('/api/groups/create', { method: 'POST', body: JSON.stringify({ name: groupName, creator_pubkey: myPubkey }) });

  }

    router.push(`/seal?pact=true&chat_id=${activeChat.id}&chat_type=${activeChat.type}`);

        <div className="flex justify-between items-center mb-4">
          <h1 className="text-xl font-bold" style={{color: 'var(--text-main)'}}>Messages</h1>
          <button onClick={() => setShowGroupModal(true)} className="bg-[#E76F51] text-white text-xs font-bold px-2 py-1 rounded-lg">+ Groupe</button>
        </div>
        {groups.length > 0 && (

                <div className="w-8 h-8 rounded-lg bg-[#4A90C2] text-white flex items-center justify-center font-bold text-sm">G</div>

              </div>
            ))}

        )}
        <h2 className="text-xs font-mono font-bold text-[#A89070] uppercase mb-2">Messages Privés</h2>
        {contacts.filter(c => c.handle !== myHandle).map(c => (

            <div className="font-bold text-sm" style={{color: 'var(--text-main)'}}>@{c.handle}</div>
          </div>
        ))}
      </div>

      {activeChat ? (
        <div className={`bg-[var(--bg-surface)] border border-[var(--border-color)] rounded-2xl flex flex-col overflow-hidden w-full md:w-2/3`}>
          <div className="p-4 border-b border-[var(--border-color)] flex justify-between items-center">

function PactModal({ pactId, onClose }: { pactId: string, onClose: () => void }) {

  useEffect(() => { loadPact(); }, [pactId]);

  async function signPact() {
    const pubkey = localStorage.getItem('sceau_pubkey');
    const res = await fetch('/api/pacts/sign', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ pact_id: pactId, pubkey }) });
    const data = await res.json();
    if (data.isNowSealed) alert("🎉 Le pacte est complet ! Il a été gravé sur la blockchain pour tous les signataires.");
    loadPact();
  }

  if (!pact) return null;
  const myHandle = localStorage.getItem('sceau_handle');
  const hasSigned = pact.signataires.includes(myHandle);
  const isSealed = pact.status === 'scelle';

  return (
    <div className="fixed inset-0 bg-black/80 z-[60] flex items-center justify-center p-4 backdrop-blur-md">
      <div className={`bg-[var(--bg-surface)] w-full max-w-md rounded-3xl p-8 border-2 ${isSealed ? 'border-[#7A9B5A]' : 'border-[#4A90C2]'}`}>

          {isSealed && <span className="bg-[#F5F9EE] text-[#5D7A3A] px-2 py-1 rounded text-xs font-bold">🔒 SCELLÉ</span>}

        {pact.file_path && (

            📁 Voir le fichier joint au pacte
          </a>
        )}

        <div className="mb-6">

          <div className="flex flex-wrap gap-2">
            {pact.signataires.map((s: string) => <span key={s} className="bg-[#F5F9EE] text-[#5D7A3A] border border-[#A8C77E] px-3 py-1 rounded-lg text-sm font-bold">✅ @{s}</span>)}

        </div>

    </div>
  );
}

  const router = useRouter();
  const [data, setData] = useState<any>(null);
  const [error, setError] = useState('');
  const [activeTab, setActiveTab] = useState('public');

  const [showPortfolioModal, setShowPortfolioModal] = useState(false);
  const [selectedSeals, setSelectedSeals] = useState<string[]>([]);
  const [filterType, setFilterType] = useState('all');
  const [startDate, setStartDate] = useState('');
  const [endDate, setEndDate] = useState('');

  async function loadProfil() {
    const pubkey = localStorage.getItem('sceau_pubkey');
    if (!pubkey) { router.push('/login'); return; }
    try {
      const res = await fetch(`/api/profil?pubkey=${encodeURIComponent(pubkey)}`);
      const d = await res.json();
      if (res.ok && d.handle) setData({ ...d, seals: d.seals || [], badges: d.badges || [], favorites: d.favorites || [] });
      else setError("Profil introuvable.");
    } catch (e) { setError("Erreur serveur."); }
  }

  useEffect(() => { loadProfil(); }, [router]);

  const openPortfolioModal = () => {
    setShowPortfolioModal(true);
    setSelectedSeals(data.seals.map((s: any) => s.id));
    setFilterType('all');
    setStartDate('');
    setEndDate('');
  };

  const applyFilters = (type: string, start: string, end: string) => {
    let filtered = data.seals;
    if (type !== 'all') {
      if (type === 'flash') filtered = filtered.filter((s: any) => s.is_flash);
      else filtered = filtered.filter((s: any) => s.visibility === type && !s.is_flash);
    }
    if (start) filtered = filtered.filter((s: any) => new Date(s.timestamp) >= new Date(start));
    if (end) filtered = filtered.filter((s: any) => new Date(s.timestamp) <= new Date(end));
    setSelectedSeals(filtered.map((s: any) => s.id));
  };

  const handleTypeChange = (type: string) => { setFilterType(type); applyFilters(type, startDate, endDate); };
  const handleStartDate = (date: string) => { setStartDate(date); applyFilters(filterType, date, endDate); };
  const handleEndDate = (date: string) => { setEndDate(date); applyFilters(filterType, startDate, date); };

  const toggleSeal = (id: string) => {
    if (selectedSeals.includes(id)) setSelectedSeals(selectedSeals.filter(sid => sid !== id));
    else setSelectedSeals([...selectedSeals, id]);
  };

  const generatePDF = () => {
    const sealsToInclude = data.seals.filter((s: any) => selectedSeals.includes(s.id));
    if (sealsToInclude.length === 0) return alert("Sélectionne au moins une œuvre !");
    generatePortfolioPDF(data.handle, sealsToInclude, startDate, endDate, data.is_pro);
    setShowPortfolioModal(false);
  };

  async function changeVisibility(sealId: string, newVisibility: string) {
    const pubkey = localStorage.getItem('sceau_pubkey');
    await fetch('/api/seal/update-visibility', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ seal_id: sealId, pubkey, visibility: newVisibility })
    });
    loadProfil();
  }

  if (error) return <div className="p-10 text-center font-mono text-red-500">{error}</div>;
  if (!data) return <div className="p-10 text-center font-mono">Chargement...</div>;

  const counts = {
    public: data.seals.filter((s:any) => s.visibility === 'public' && !s.is_flash).length,
    friends: data.seals.filter((s:any) => s.visibility === 'friends' && !s.is_flash).length,
    private: data.seals.filter((s:any) => s.visibility === 'private' && !s.is_flash).length,
    flash: data.seals.filter((s:any) => s.is_flash).length,
  };

  // ONGLET FAVORIS

    return (

        <button onClick={() => setActiveTab('public')} className="text-sm font-bold text-[#A89070] mb-4 hover:underline">← Retour au coffre</button>

  async function startRecording() {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      const mediaRecorder = new MediaRecorder(stream);
      mediaRecorderRef.current = mediaRecorder;
      audioChunksRef.current = [];
      mediaRecorder.ondataavailable = (event) => { if (event.data.size > 0) audioChunksRef.current.push(event.data); };
      mediaRecorder.onstop = () => {
        const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
        setFile(new File([audioBlob], `audio-${Date.now()}.webm`, { type: 'audio/webm' }));
        setAudioUrl(URL.createObjectURL(audioBlob));
      };
      mediaRecorder.start(); setIsRecording(true);
    } catch (err) { alert("Impossible d'accéder au micro."); }
  }

  function stopRecording() {
    if (mediaRecorderRef.current && isRecording) {
      mediaRecorderRef.current.stop();
      mediaRecorderRef.current.stream.getTracks().forEach(track => track.stop());
      setIsRecording(false);
    }

  async function handleSeal() {
    if (!title) return setStatus("Un titre est requis");
    if (!isFlashMode && mode === 'file' && !file) return setStatus("Choisis un fichier");
    if (!isFlashMode && mode === 'text' && !textContent) return setStatus("Écris quelque chose");
    if (!isFlashMode && mode === 'media' && !file) return setStatus("Prends une photo ou vidéo");
    if (!isFlashMode && mode === 'audio' && !file) return setStatus("Enregistre un audio");

    try {
      if (isFlashMode && !file) {

        let hashStr = ''; for (let i = 0; i < hash.length; i++) hashStr += String.fromCharCode(hash[i]);
        return sendToServer(btoa(hashStr), null);
      }

      let fileToSeal: File;
      
      if (mode === 'text') {
        const blob = new Blob([textContent], { type: 'text/plain' });

      } else {
        if (!file) return setStatus("Fichier manquant");
        fileToSeal = file;
      }

      const buffer = await fileToSeal.arrayBuffer();
      const hash = sha256(new Uint8Array(buffer));
      let hashStr = ''; for (let i = 0; i < hash.length; i++) hashStr += String.fromCharCode(hash[i]);
      
      sendToServer(btoa(hashStr), fileToSeal);

        }

      } else {

'use client';
import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { ed25519 } from '@noble/curves/ed25519';
import Link from 'next/link';

export default function Signup() {
  const router = useRouter();
  const [handle, setHandle] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [showPassword, setShowPassword] = useState(false);
  const [status, setStatus] = useState('');

  async function handleSignup() {
    if (!handle || !email || !password) return setStatus("Tous les champs sont requis");
    setStatus("Création en cours...");
    
    try {
      const secret = crypto.getRandomValues(new Uint8Array(32));
      const pubkey = ed25519.getPublicKey(secret);
      let pubkeyStr = ''; for (let i = 0; i < pubkey.length; i++) pubkeyStr += String.fromCharCode(pubkey[i]);
      let secretStr = ''; for (let i = 0; i < secret.length; i++) secretStr += String.fromCharCode(secret[i]);
      const pubkeyB64 = btoa(pubkeyStr);

      localStorage.setItem('sceau_secret', btoa(secretStr));

      const res = await fetch('/api/auth/signup', { 
        method: 'POST', 
        headers: { 'Content-Type': 'application/json' }, 
        body: JSON.stringify({ handle, email, password, pubkey: pubkeyB64 }) 
      });
      
      const data = await res.json();
      
      if (res.ok) {
        localStorage.setItem('sceau_pubkey', pubkeyB64);
        localStorage.setItem('sceau_handle', handle);
        setStatus("✅ Compte créé !");
        setTimeout(() => router.push('/feed'), 500);
      } else {
        setStatus("❌ " + (data.error || "Une erreur est survenue"));
      }
    } catch (e) { setStatus("❌ Impossible de contacter le serveur. Vérifiez votre connexion."); }
  }

  const handleKeyPress = (e: React.KeyboardEvent) => { if (e.key === 'Enter') handleSignup(); };

  return (
    <div className="max-w-md mx-auto bg-[var(--bg-surface)] p-8 rounded-3xl border border-[var(--border-color)] shadow-sm mt-10">
      <h1 className="text-3xl font-bold mb-6 font-serif" style={{color: 'var(--text-main)'}}>Créer un compte</h1>
      
      <label className="block text-xs font-bold text-[var(--text-muted)] uppercase mb-1 font-mono">Pseudo</label>
      <input type="text" onKeyDown={handleKeyPress} className="w-full border-2 border-[var(--border-color)] p-3 rounded-xl mb-4 bg-transparent" style={{color: 'var(--text-main)'}} value={handle} onChange={(e) => setHandle(e.target.value)} />

      <label className="block text-xs font-bold text-[var(--text-muted)] uppercase mb-1 font-mono">Mot de passe</label>

      fetch('/api/verify/pulse').then(res => res.json()).then(data => {
        setPulseData(data.pulse || []);
        setPulseLoading(false);
      });
    };
    fetchPulse();
    const interval = setInterval(fetchPulse, 5000);
    return () => clearInterval(interval);
  }, [tab]);

  async function verifyFile(file: File) {
    setLoading(true);
    const buffer = await file.arrayBuffer();
    const hash = sha256(new Uint8Array(buffer));
    let hashStr = ''; for (let i = 0; i < hash.length; i++) hashStr += String.fromCharCode(hash[i]);
    const res = await fetch('/api/verify', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ hash: btoa(hashStr) }) });
    const data = await res.json(); setResult(data); setLoading(false);
  }

  async function verifyHash(input: string) {
    if(!input) return; setLoading(true);
    const res = await fetch('/api/verify', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ hash: input }) });
    const data = await res.json(); setResult(data); setLoading(false);
  }

  async function searchByTitle() {
    if(!titleQuery) return; setLoading(true);

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';
import crypto from 'crypto';
import { Resend } from 'resend';

export async function POST(req: Request) {
  const { email } = await req.json();
  if (!email) return NextResponse.json({ error: 'Email requis' }, { status: 400 });

  const user = await pool.query("SELECT id FROM users WHERE email = $1", [email]);
  if (user.rows.length === 0) return NextResponse.json({ error: 'Email introuvable' }, { status: 404 });

  const token = crypto.randomBytes(32).toString('hex');
  const expiry = new Date(Date.now() + 3600000); 

  await pool.query("UPDATE users SET reset_token = $1, reset_expiry = $2 WHERE email = $3", [token, expiry, email]);

  const resetLink = `${process.env.NEXT_PUBLIC_APP_URL || 'http://localhost:3000'}/reset-password?token=${token}`;
  
  try {
    const resendKey = process.env.RESEND_API_KEY;
    if (!resendKey) {
      // Si pas de clé, on simule l'envoi pour le développement
      console.log("⚠️ Pas de clé Resend. Lien de test :", resetLink);
      return NextResponse.json({ success: true, message: "Lien généré (voir console)" });
    }

    const resend = new Resend(resendKey);
    await resend.emails.send({
      from: 'SCEAU <onboarding@resend.dev>',
      to: email,
      subject: "Réinitialisation de votre mot de passe",
      html: `<p>Bonjour,</p><p>Voici votre lien : <a href="${resetLink}">Réinitialiser mon mot de passe</a></p>`,
    });
    return NextResponse.json({ success: true, message: "Email envoyé !" });
  } catch (error) {
    return NextResponse.json({ error: "Erreur d'envoi d'email" }, { status: 500 });
  }
}

import { pool } from '@/db/pool';
import bcrypt from 'bcryptjs';

export async function POST(req: Request) {
  try {
    const { email, password } = await req.json();

    const user = await pool.query("SELECT * FROM users WHERE email = $1", [email]);
    if (user.rows.length === 0) {
      return NextResponse.json({ error: 'Email ou mot de passe incorrect' }, { status: 401 });
    }

    const isValid = await bcrypt.compare(password, user.rows[0].password_hash);
    if (!isValid) {
      return NextResponse.json({ error: 'Email ou mot de passe incorrect' }, { status: 401 });
    }

    // On renvoie le handle et la clé publique (convertie en base64)
    return NextResponse.json({
      handle: user.rows[0].handle,
      pubkey: Buffer.from(user.rows[0].pubkey).toString('base64')
    });
  } catch (error) {
    return NextResponse.json({ error: 'Erreur serveur' }, { status: 500 });
  }
}

import { pool } from '@/db/pool';

export async function POST(req: Request) {
  const { token, newPassword } = await req.json();
  if (!token || !newPassword) return NextResponse.json({ error: 'Champs requis' }, { status: 400 });

  const user = await pool.query("SELECT id FROM users WHERE reset_token = $1 AND reset_expiry > NOW()", [token]);
  if (user.rows.length === 0) return NextResponse.json({ error: 'Lien invalide ou expiré' }, { status: 400 });

  const salt = await bcrypt.genSalt(10);
  const password_hash = await bcrypt.hash(newPassword, salt);

  await pool.query("UPDATE users SET password_hash = $1, reset_token = NULL, reset_expiry = NULL WHERE id = $2", [password_hash, user.rows[0].id]);

  return NextResponse.json({ success: true });
}

      return NextResponse.json({ error: 'Ce pseudo ou cet email est déjà pris' }, { status: 409 });

    );
    return NextResponse.json({ success: true });

import { pool } from '@/db/pool';

export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const myHandle = searchParams.get('me');

  if (!myHandle) return NextResponse.json({ friends_seals: [] });

  try {
    // 1. On trouve l'ID de l'utilisateur
    const me = await pool.query("SELECT id FROM users WHERE handle = $1", [myHandle]);
    if (me.rows.length === 0) return NextResponse.json({ friends_seals: [] });
    const myId = me.rows[0].id;

    // 2. On trouve les gens avec qui j'ai discuté (mes contacts)
    const contactsQuery = await pool.query(`
      SELECT DISTINCT receiver_id as contact_id FROM messages WHERE sender_id = $1
      UNION
      SELECT DISTINCT sender_id as contact_id FROM messages WHERE receiver_id = $1
    `, [myId]);

    if (contactsQuery.rows.length === 0) return NextResponse.json({ friends_seals: [] });
    const contactIds = contactsQuery.rows.map(r => r.contact_id);

    // 3. On récupère leurs dernières œuvres publiques (5 max)
    const seals = await pool.query(`
      SELECT s.id, s.title, s.timestamp, u.handle as author
      FROM seals s JOIN users u ON s.author_id = u.id
      WHERE s.author_id = ANY($1) AND s.is_public = TRUE
      ORDER BY s.timestamp DESC LIMIT 5
    `, [contactIds]);

    return NextResponse.json({ friends_seals: seals.rows });
  } catch (error) { 
    return NextResponse.json({ error: 'Erreur' }, { status: 500 }); 
  }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';

export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const myHandle = searchParams.get('me');
  const category = searchParams.get('category');

  try {
    let params: any[] = [];
    let whereClause = "s.visibility = 'public'";

    if (category && category !== 'all') {
      params.push(category);
      whereClause += ` AND s.category = $${params.length}`;
    }

    if (myHandle) {
      const me = await pool.query("SELECT id FROM users WHERE handle = $1", [myHandle]);
      if (me.rows.length > 0) {
        const myId = me.rows[0].id;
        const friendsRes = await pool.query(`
          SELECT requester_id as id FROM friendships WHERE addressee_id = $1 AND status = 'accepted'
          UNION
          SELECT addressee_id as id FROM friendships WHERE requester_id = $1 AND status = 'accepted'
        `, [myId]);
        const friendIds = friendsRes.rows.map(r => r.id);
        if (friendIds.length > 0) {
          params.push(friendIds);
          whereClause = `(s.visibility = 'public' OR (s.visibility = 'friends' AND s.author_id = ANY($${params.length})))`;
          if (category && category !== 'all') {
             whereClause += ` AND s.category = $${params.length - 1}`; // On garde le filtre catégorie
          }
        }
      }
    }

    const sql = `
      SELECT s.id, 
        CASE WHEN s.is_mystery = true THEN '🔒 Sceau Mystère' ELSE s.title END as title,
        CASE WHEN s.is_mystery = true AND (SELECT COUNT(*) FROM likes WHERE seal_id = s.id) < s.unlock_likes THEN null ELSE s.file_path END as file_path,
        s.is_mystery, s.unlock_likes, s.is_flash, s.ots_proof,
        s.description, s.category, s.timestamp, s.visibility, encode(s.hash, 'base64') as hash, u.handle as author,
        s.project_id, s.version_number,
        (SELECT COUNT(*) FROM likes WHERE seal_id = s.id AND reaction_type = 'like') as likes_like,
        (SELECT COUNT(*) FROM likes WHERE seal_id = s.id AND reaction_type = 'wow') as likes_wow,
        (SELECT COUNT(*) FROM likes WHERE seal_id = s.id AND reaction_type = 'inspired') as likes_inspired
      FROM seals s JOIN users u ON s.author_id = u.id
      WHERE ${whereClause} ORDER BY s.timestamp DESC LIMIT 30
    `;
    
    const result = await pool.query(sql, params);
    return NextResponse.json({ seals: result.rows });
  } catch (error) { return NextResponse.json({ error: 'Erreur' }, { status: 500 }); }
}

    if (!me) return NextResponse.json({ error: 'Non connecté' }, { status: 401 });
    // NOUVEAU : On nettoie le pseudo au cas où il y aurait un @
    me = me.replace('@', '');

    const myIdRes = await pool.query("SELECT id FROM users WHERE handle = $1", [me]);
    if (myIdRes.rows.length === 0) {
      // Message plus clair pour le débogage

    }
    const myId = myIdRes.rows[0].id;

    if (search) {
      const users = await pool.query(`
        SELECT handle, role FROM users 
        WHERE handle ILIKE $1 AND id != $2 LIMIT 10
      `, [`%${search.replace('@', '')}%`, myId]);
      return NextResponse.json({ searchResults: users.rows });
    }

    const friendsQuery = await pool.query(`
      SELECT 
        u.id, 
        u.handle,
        (SELECT title FROM seals WHERE author_id = u.id AND visibility = 'public' ORDER BY created_at DESC LIMIT 1) as last_seal,
        (SELECT id FROM seals WHERE author_id = u.id AND visibility = 'public' ORDER BY created_at DESC LIMIT 1) as last_seal_id
      FROM friendships f
      JOIN users u ON u.id = CASE WHEN f.requester_id = $1 THEN f.addressee_id ELSE f.requester_id END
      WHERE (f.requester_id = $1 OR f.addressee_id = $1) AND f.status = 'accepted'
    `, [myId]);

    const receivedQuery = await pool.query(`
      SELECT u.handle FROM friendships f 
      JOIN users u ON u.id = f.requester_id
      WHERE f.addressee_id = $1 AND f.status = 'pending'
    `, [myId]);

    return NextResponse.json({ 
      friends: friendsQuery.rows, 
      received: receivedQuery.rows 
    });

  } catch (error: any) {
    console.error("Erreur API Friends:", error);
    return NextResponse.json({ error: "Erreur base de données : " + error.message }, { status: 500 });

    // 2. Récupérer les derniers commits publics via l'API GitHub
    const ghRes = await fetch(`https://api.github.com/users/${github_username}/events/public`, {
      headers: { 'User-Agent': 'SCEAU-App' }
    });
    const events = await ghRes.json();
    
    let commitsScelles = 0;
    for (const event of events) {
      if (event.type === 'PushEvent' && event.payload.commits.length > 0) {
        for (const commit of event.payload.commits) {
          // On crée un hash basé sur le message et l'ID du commit
          const hashStr = `${commit.sha}-${commit.message}`;
          const hash = sha256(new TextEncoder().encode(hashStr));
          let hashB64 = btoa(String.fromCharCode(...hash));

          // On insère dans la base (ON CONFLICT DO NOTHING pour éviter les doublons)
          await pool.query(
            `INSERT INTO seals (hash, timestamp, author_id, title, category, work_type, is_public) 
             VALUES (decode($1, 'base64'), $2, $3, $4, 'Code', 'commit', true) ON CONFLICT DO NOTHING`,
            [hashB64, event.created_at, userId, `Commit: ${commit.message.slice(0, 50)}`]
          );
          commitsScelles++;
        }
      }
    }
    return NextResponse.json({ success: true, commitsScelles });
  } catch (error) {
    return NextResponse.json({ error: 'Erreur GitHub' }, { status: 500 });
  }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';
import { sha256 } from '@noble/hashes/sha256';
import { submitToOts } from '@/lib/ots';

export async function POST(req: Request) {
  const { pact_id, pubkey } = await req.json();
  const user = await pool.query("SELECT id FROM users WHERE encode(pubkey, 'base64') = $1", [pubkey]);
  if (user.rows.length === 0) return NextResponse.json({ error: 'Inconnu' }, { status: 401 });

  try {
    await pool.query(
      "INSERT INTO pact_signatures (pact_id, user_id) VALUES ($1, $2) ON CONFLICT DO NOTHING",
      [pact_id, user.rows[0].id]
    );

    const pact = await pool.query(
      "SELECT title, description, required_signatures, status FROM pacts WHERE id = $1",
      [pact_id]
    );
    const sigs = await pool.query(
      `SELECT u.handle, s.created_at FROM pact_signatures s JOIN users u ON s.user_id = u.id
       WHERE s.pact_id = $1 ORDER BY s.created_at ASC`,
      [pact_id]
    );

    const required = pact.rows[0].required_signatures;
    const current = sigs.rows.length;

    if (current >= required && pact.rows[0].status !== 'scelle') {
      const canonical = JSON.stringify({
        title: pact.rows[0].title,
        description: pact.rows[0].description || '',
        signers: sigs.rows.map((r: any) => ({ handle: r.handle, signed_at: new Date(r.created_at).toISOString() })),
      });
      const pactHash = sha256(new TextEncoder().encode(canonical));

      let otsProof: string | null = null;
      try {
        const ots = await submitToOts(new Uint8Array(pactHash));
        if (ots.ok && ots.proof) otsProof = ots.proof.toString('base64');
      } catch (e) { console.error('[Pact OTS]', e); }

      await pool.query(
        `UPDATE pacts SET status='scelle', scelle_at=NOW(), pact_hash=$1, ots_proof=$2 WHERE id=$3`,
        [Buffer.from(pactHash), otsProof, pact_id]
      );
    }

    return NextResponse.json({ success: true, current, required });
  } catch (e) {
    console.error(e);
    return NextResponse.json({ error: 'Erreur' }, { status: 500 });
  }
}

    const userId = user.rows[0].id;

    `, [userId]);

    let cursor = new Date();
    cursor.setUTCHours(0, 0, 0, 0);
    if (!uniqueDatesSet.has(cursor.toISOString().split('T')[0])) {
      cursor.setDate(cursor.getDate() - 1);
    }
    while (uniqueDatesSet.has(cursor.toISOString().split('T')[0])) {
      streak++;
      cursor.setDate(cursor.getDate() - 1);

  const { id } = await params;
  try {
    const res = await pool.query("SELECT hash FROM seals WHERE id = $1", [id]);
    if (res.rows.length === 0) return NextResponse.json({ error: 'Introuvable' }, { status: 404 });

    const upgraded = await upgradeOts(new Uint8Array(res.rows[0].hash));
    if (!upgraded) {
      return NextResponse.json({ status: 'pending', message: 'Pas encore confirmé. Réessaie dans quelques heures.' });
    }
    await pool.query(
      "UPDATE seals SET ots_proof = $1, ots_upgraded_at = NOW() WHERE id = $2",
      [upgraded.toString('base64'), id]
    );
    return NextResponse.json({ status: 'upgraded', size: upgraded.length });
  } catch {
    return NextResponse.json({ error: 'Erreur' }, { status: 500 });
  }
}

      status: 200,
      headers: {
        'Content-Type': 'application/octet-stream',
        'Content-Disposition': `attachment; filename="${safeTitle}.ots"`,
      },
    });
  } catch {
    return NextResponse.json({ error: 'Erreur' }, { status: 500 });
  }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';
import { r2Client } from '@/lib/r2';
import { DeleteObjectCommand } from '@aws-sdk/client-s3';

export async function DELETE(req: Request) {
  try {
    const { searchParams } = new URL(req.url);
    const sealId = searchParams.get('id');
    const pubkey = searchParams.get('pubkey');

    if (!sealId || !pubkey) {
      return NextResponse.json({ error: 'ID ou clé manquant' }, { status: 400 });
    }

    const user = await pool.query("SELECT id FROM users WHERE encode(pubkey, 'base64') = $1", [pubkey]);
    if (user.rows.length === 0) return NextResponse.json({ error: 'Inconnu' }, { status: 401 });
    
    const seal = await pool.query("SELECT author_id, file_path FROM seals WHERE id = $1", [sealId]);
    if (seal.rows.length === 0) return NextResponse.json({ error: 'Sceau introuvable' }, { status: 404 });
    
    if (seal.rows[0].author_id !== user.rows[0].id) {
      return NextResponse.json({ error: 'Non autorisé' }, { status: 403 });
    }

    // CORRECTION : On supprime le fichier de Cloudflare R2, pas du disque local
    if (seal.rows[0].file_path) {
      try {
        const filename = seal.rows[0].file_path.split('/').pop();
        await r2Client.send(new DeleteObjectCommand({
          Bucket: process.env.S3_BUCKET,
          Key: filename
        }));
      } catch (e) { console.log("Fichier introuvable sur R2 ou déjà supprimé"); }
    }

    await pool.query("DELETE FROM seals WHERE id = $1", [sealId]);

    return NextResponse.json({ success: true });
  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Erreur serveur' }, { status: 500 });
  }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';
import { r2Client } from '@/lib/r2';
import { PutObjectCommand } from '@aws-sdk/client-s3';
import { submitToOts } from '@/lib/ots';

export async function POST(req: Request) {
  try {
    const data = await req.formData();
    const hash = data.get('hash') as string;
    const title = data.get('title') as string;
    const description = data.get('description') as string;
    const isFlash = data.get('isFlash') === 'true';
    const isMystery = data.get('isMystery') === 'true';
    const isEphemeral = data.get('isEphemeral') === 'true';
    const unlockLikes = parseInt(data.get('unlockLikes') as string || '0');
    const pubkey = data.get('pubkey') as string;
    const file = data.get('file') as File;

    const user = await pool.query("SELECT id FROM users WHERE encode(pubkey, 'base64') = $1", [pubkey]);
    if (user.rows.length === 0) return NextResponse.json({ error: 'Utilisateur inconnu' }, { status: 401 });
    const authorId = user.rows[0].id;

    let filePath: string | null = null;
    let fileSize = 0;
    let otsProof: string | null = null;
    let otsCalendars: string[] = [];

    try {
      const hashBytes = new Uint8Array(Buffer.from(hash, 'base64'));
      const ots = await submitToOts(hashBytes);
      if (ots.ok && ots.proof) {
        otsProof = ots.proof.toString('base64');
        otsCalendars = ots.calendars;
        console.log(`[OTS] ancré via ${ots.calendars.length} calendar(s)`);
      } else {
        console.warn('[OTS] échec :', ots.error);
      }
    } catch (e) {
      console.error('[OTS] exception :', e);
    }

    if (!isFlash && file) {
      const quotaCheck = await pool.query(
        "SELECT COALESCE(SUM(file_size), 0) as used FROM seals WHERE author_id = $1 AND is_hosted = true",
        [authorId]
      );
      const currentUsage = parseInt(quotaCheck.rows[0].used);
      if (currentUsage + file.size > 50 * 1024 * 1024) {
        return NextResponse.json({ error: 'Quota de 50 Mo dépassé.' }, { status: 403 });
      }

      fileSize = file.size;
      const buffer = Buffer.from(await file.arrayBuffer());
      const filename = `${hash.replace(/[^a-zA-Z0-9]/g, '').slice(0, 20)}-${file.name}`;
      await r2Client.send(new PutObjectCommand({
        Bucket: process.env.S3_BUCKET,
        Key: filename,
        Body: buffer,
        ContentType: file.type,
      }));
      filePath = `${process.env.AWS_ENDPOINT_URL_S3}/${process.env.S3_BUCKET}/${filename}`;
    }

    const result = await pool.query(
      `INSERT INTO seals (
        hash, timestamp, author_id, title, description, is_public, visibility,
        is_hosted, file_path, file_size, is_flash, is_mystery, unlock_likes, is_ephemeral,
        ots_proof, ots_submitted_at
      ) VALUES (
        decode($1, 'base64'), NOW(), $2, $3, $4, true, 'public',
        $5, $6, $7, $8, $9, $10, $11, $12, $13
      ) RETURNING id`,
      [hash, authorId, title, description,
       !isFlash && filePath ? true : false, filePath, fileSize,
       isFlash, isMystery, unlockLikes, isEphemeral,
       otsProof, otsProof ? new Date() : null]
    );

    return NextResponse.json({
      seal_id: result.rows[0].id,
      ots_secured: otsProof !== null,
      ots_calendars: otsCalendars,
    });
  } catch (error: any) {
    console.error(error);
    return NextResponse.json({ error: error.message || 'Erreur serveur' }, { status: 500 });
  }
}

    );

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';

export async function GET(req: Request, { params }: { params: Promise<{ handle: string }> }) {
  const { handle } = await params;
  const { searchParams } = new URL(req.url);
  const viewerHandle = searchParams.get('viewer');

  try {
    const user = await pool.query("SELECT id, handle FROM users WHERE handle = $1", [handle]);
    if (user.rows.length === 0) return NextResponse.json({ error: 'Introuvable' }, { status: 404 });
    const targetUserId = user.rows[0].id;

    let relationship = 'none';
    let viewerId = null;

    if (viewerHandle) {
      const viewer = await pool.query("SELECT id FROM users WHERE handle = $1", [viewerHandle]);
      if (viewer.rows.length > 0) {
        viewerId = viewer.rows[0].id;
        const friend = await pool.query(
          "SELECT status FROM friendships WHERE (requester_id = $1 AND addressee_id = $2) OR (requester_id = $2 AND addressee_id = $1)",
          [viewerId, targetUserId]
        );
        if (friend.rows.length > 0) {
          if (friend.rows[0].status === 'accepted') relationship = 'friend';
          else relationship = 'pending';
        }
      }
    }

    let visFilter = "s.visibility = 'public'";
    if (viewerId === targetUserId) visFilter = "1=1"; 
    else if (relationship === 'friend') visFilter = "s.visibility IN ('public', 'friends')";

    const seals = await pool.query(`
      SELECT id, title, description, timestamp, is_hosted, file_path, visibility 
      FROM seals s WHERE author_id = $1 AND ${visFilter} ORDER BY created_at DESC
    `, [targetUserId]);

    return NextResponse.json({ handle: user.rows[0].handle, relationship, seals: seals.rows });
  } catch (error) { return NextResponse.json({ error: 'Erreur' }, { status: 500 }); }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';

export async function POST(req: Request) {
  try {
    const { hash, query } = await req.json();
    
    // 1. Recherche par Hash (Vérification classique)
    if (hash) {
      const result = await pool.query(`
        SELECT s.title, s.timestamp, s.is_hosted, s.file_path, s.category, u.handle as author
        FROM seals s JOIN users u ON s.author_id = u.id
        WHERE encode(s.hash, 'base64') = $1
        ORDER BY s.timestamp ASC
      `, [hash]);
      if (result.rows.length > 0) return NextResponse.json({ found: true, seals: result.rows });
      return NextResponse.json({ found: false });
    }

    // 2. Recherche par Titre (Découverte publique)
    if (query) {
      const result = await pool.query(`
        SELECT s.title, s.timestamp, s.is_hosted, s.file_path, s.category, encode(s.hash, 'base64') as hash, u.handle as author
        FROM seals s JOIN users u ON s.author_id = u.id
        WHERE s.title ILIKE $1 AND s.is_public = TRUE
        ORDER BY s.timestamp DESC LIMIT 20
      `, [`%${query}%`]);
      
      return NextResponse.json({ found: result.rows.length > 0, seals: result.rows });
    }

    return NextResponse.json({ error: 'Requête invalide' }, { status: 400 });
  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Erreur' }, { status: 500 });
  }
}
