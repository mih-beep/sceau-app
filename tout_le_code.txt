import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';

export async function POST(req: Request) {
  try {
    const { handle, pubkey } = await req.json();

    // 1. Vérification des données
    if (!handle || !pubkey) {
      return NextResponse.json({ error: 'Pseudo et clé requis' }, { status: 400 });
    }

    // 2. Insertion
    const result = await pool.query(
      "INSERT INTO users (handle, pubkey) VALUES ($1, decode($2, 'base64')) RETURNING id, handle", 
      [handle, pubkey]
    );
    
    return NextResponse.json(result.rows[0]);

  } catch (error: any) {
    // Si le pseudo existe déjà (Code PostgreSQL 23505 = violation d'unicité)
    if (error.code === '23505') {
      return NextResponse.json({ error: 'Ce pseudo est déjà pris' }, { status: 409 });
    }
    console.error(error);
    return NextResponse.json({ error: 'Erreur serveur lors de l\'inscription' }, { status: 500 });
  }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';

export async function POST(req: Request) {
  try {
    // On récupère TOUTES les nouvelles données depuis l'interface
    const { hash, title, pubkey, category, isHosted, filePath } = await req.json();

    if (!hash || !title || !pubkey) {
      return NextResponse.json({ error: 'Données manquantes' }, { status: 400 });
    }

    // 1. Trouver l'auteur
    const user = await pool.query(
      "SELECT id FROM users WHERE encode(pubkey, 'base64') = $1", 
      [pubkey]
    );
    
    if (user.rows.length === 0) {
      return NextResponse.json({ error: 'Utilisateur introuvable. Inscris-toi d\'abord.' }, { status: 401 });
    }

    // 2. Appeler le protocole (ton stub)
    const protocolRes = await fetch('http://localhost:8080/v1/seal', { 
      method: 'POST', 
      headers: { 'Content-Type': 'application/json' }, 
      body: JSON.stringify({ hash, title }) 
    });
    
    if (!protocolRes.ok) throw new Error('Le protocole a échoué');
    const protocolData = await protocolRes.json();

    // 3. Sauvegarder en base
    await pool.query(
      `INSERT INTO seals (hash, timestamp, author_id, title, category, is_hosted, file_path) 
       VALUES (decode($1, 'base64'), $2, $3, $4, $5, $6, $7)`,
      [hash, protocolData.timestamp, user.rows[0].id, title, category || 'Art', isHosted || false, filePath || null]
    );

    return NextResponse.json(protocolData);

  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Erreur lors du scellement' }, { status: 500 });
  }
}

import { NextResponse } from 'next/server';
import { pool } from '@/db/pool';

export async function POST(req: Request) {
  try {
    const { hash } = await req.json();
    
    if (!hash) {
      return NextResponse.json({ error: 'Hash manquant' }, { status: 400 });
    }

    const result = await pool.query(`
      SELECT s.title, s.timestamp, s.is_hosted, s.file_path, s.category, u.handle as author
      FROM seals s 
      JOIN users u ON s.author_id = u.id
      WHERE encode(s.hash, 'base64') = $1
    `, [hash]);

    if (result.rows.length > 0) {
      return NextResponse.json({ found: true, seal: result.rows[0] });
    }
    
    return NextResponse.json({ found: false });
    
  } catch (error) {
    console.error(error);
    return NextResponse.json({ error: 'Erreur lors de la vérification' }, { status: 500 });
  }
}

import { useState } from 'react';
import { ed25519 } from '@noble/curves/ed25519';

export default function Signup() {
  const [handle, setHandle] = useState('');
  const [status, setStatus] = useState('');

  async function handleSignup() {
    if (!handle) { setStatus("Pseudo requis"); return; }
    setStatus("Création...");
    try {
      const secret = crypto.getRandomValues(new Uint8Array(32));
      const pubkey = ed25519.getPublicKey(secret);
      let pubkeyStr = ''; for (let i = 0; i < pubkey.length; i++) pubkeyStr += String.fromCharCode(pubkey[i]);
      let secretStr = ''; for (let i = 0; i < secret.length; i++) secretStr += String.fromCharCode(secret[i]);
      localStorage.setItem('sceau_secret', btoa(secretStr));
      localStorage.setItem('sceau_pubkey', btoa(pubkeyStr));

      const res = await fetch('/api/signup', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ handle, pubkey: btoa(pubkeyStr) }) });
      if (res.ok) setStatus("✅ Compte créé ! Vous pouvez sceller.");
      else { const data = await res.json(); setStatus("Erreur : " + data.error); }
    } catch (e) { setStatus("Erreur réseau"); }
  }

  return (
    <div className="max-w-md mx-auto bg-[#FFFCF7] p-8 rounded-3xl border border-[#EFE0CA] shadow-sm mt-10">
      <h1 className="text-3xl font-bold mb-2">Créer un identifiant</h1>
      <p className="text-[#7A5A3A] mb-8 italic" style={{fontFamily: 'Cormorant Garamond'}}>Anonyme, gratuit, et unique.</p>
      <label className="block text-xs font-bold text-[#A89070] uppercase mb-1 mono">Pseudo</label>
      <input type="text" placeholder="Ex: marie" className="w-full border-2 border-[#EFE0CA] focus:border-[#E76F51] outline-none p-3 rounded-xl text-[#2C1810] bg-white transition mb-6" value={handle} onChange={(e) => setHandle(e.target.value)} />
      <button onClick={handleSignup} className="w-full bg-gradient-to-r from-[#E76F51] to-[#F4A261] text-white p-4 rounded-xl font-bold shadow-md hover:shadow-lg transition">
        Créer mon identité
      </button>
      {status && <p className="mt-4 font-bold text-center text-[#C44A2E]">{status}</p>}
    </div>
  );
}

import { useState } from 'react';
import { sha256 } from '@noble/hashes/sha256';

export default function Verify() {
  const [tab, setTab] = useState('file');
  const [result, setResult] = useState<any>(null);
  const [loading, setLoading] = useState(false);

  async function verifyFile(file: File) {
    setLoading(true);
    const buffer = await file.arrayBuffer();
    const hash = sha256(new Uint8Array(buffer));
    let hashStr = ''; for (let i = 0; i < hash.length; i++) hashStr += String.fromCharCode(hash[i]);
    const res = await fetch('/api/verify', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ hash: btoa(hashStr) }) });
    const data = await res.json(); setResult(data); setLoading(false);
  }

  async function verifyHash(input: string) {
    if(!input) return; setLoading(true);
    const res = await fetch('/api/verify', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ hash: input }) });
    const data = await res.json(); setResult(data); setLoading(false);
  }

  return (
    <div className="p-6">
      <h1 className="text-3xl font-bold font-serif mb-2 text-center">Vérifier.</h1>
      <p className="text-[#7A5A3A] italic font-serif mb-6 text-center">Prouvez l'authenticité.</p>
      <div className="flex bg-[#F9EEDC] p-1 rounded-xl mb-6">

      ) : (
        <div className="space-y-4">
          <input type="text" placeholder="Collez le hash (ex: b2e7...)" className="w-full border-2 border-[#EFE0CA] p-3 rounded-xl outline-none font-mono" onChange={e=>verifyHash(e.target.value)} />
        </div>
      )}
      {loading && <p className="text-center mt-4">Recherche...</p>}
      {result && (
        <div className={`mt-6 p-6 rounded-2xl border-2 ${result.found ? 'bg-[#F5F9EE] border-[#7A9B5A]' : 'bg-[#FDF4F0] border-[#D64545]'}`}>
          {result.found ? (

              <div className="text-3xl mb-2">✅</div>
